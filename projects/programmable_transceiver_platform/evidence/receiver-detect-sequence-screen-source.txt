"""Staged two-leg probe controller over persistent RC state; not yet chip-connected."""
import math,json,hashlib
from pathlib import Path
from release_screen import Leg
from load_screen import classify

class Probe:
    def __init__(self,loads):
        if len(loads)!=2:raise ValueError('Two conductors required')
        self.legs=[Leg(p) for p in loads];self.time=0.;self.state='idle'
        self.drive=None;self.next_event=math.inf;self.epoch=0;self.result=None
    def start(self,time,epoch,tx_busy=False):
        if tx_busy or self.state not in ('idle','ready') or epoch!=self.epoch:
            raise ValueError('Probe ownership/epoch conflict')
        if not math.isfinite(time) or time<self.time:raise ValueError('Invalid probe time')
        # Advance physical state before checking baseline, but never enable a
        # stimulus on a failed preflight.
        self.advance(time)
        if max(abs(v) for leg in self.legs for v in leg.x)>.005:
            raise ValueError('Probe baseline not settled')
        self.baseline=[leg.x[0] for leg in self.legs];self.samples=[[],[]]
        self.state='probing';self.result=None;self.drive=.2;self.started=time
        self.next_event=time+100e-9
    def advance(self,time):
        if not math.isfinite(time) or time<self.time:raise ValueError('Invalid probe time')
        while self.next_event<=time:
            when=self.next_event
            for leg in self.legs:leg.advance(when,self.drive)
            self.time=when
            if self.state=='probing':
                for i,leg in enumerate(self.legs):self.samples[i].append(float((leg.x[0]-self.baseline[i])/.2))
                if len(self.samples[0])==1:self.next_event=self.started+200e-9
                else:
                    decisions=[classify(v) for v in self.samples]
                    self.pending=decisions[0] if decisions[0]==decisions[1] else 'indeterminate'
                    self.state='releasing';self.drive=0.;self.release_start=when;self.next_event=when+1e-6
            elif self.state=='releasing':
                settled=max(abs(v) for leg in self.legs for v in leg.x)<=.005
                if settled:
                    self.result=dict(epoch=self.epoch,decision=self.pending,samples=self.samples,ready_at=when)
                    self.drive=None;self.state='ready';self.next_event=math.inf
                elif when-self.release_start>=1e-3:
                    self.drive=None;self.state='fault';self.result=None;self.next_event=math.inf
                else:self.next_event=when+1e-6
        for leg in self.legs:leg.advance(time,self.drive)
        self.time=time
    def read(self,epoch):
        if epoch!=self.epoch or self.state!='ready':raise ValueError('No current completed detection')
        import copy
        return copy.deepcopy(self.result)
    def abort(self,time):
        self.advance(time)
        self.drive=None;self.state='aborted';self.result=None;self.next_event=math.inf;self.epoch+=1


def load(rt):return dict(cp=5e-12,cr=5e-12,cc=47e-9,rs=470.,rt=rt)
def reject(fn):
    try:fn()
    except ValueError:return
    raise AssertionError('Invalid operation accepted')


def main():
    rows=[]
    for terms,expected in [((75,75),'present'),((1e8,1e8),'absent'),((75,1e8),'indeterminate')]:
        p=Probe([load(r) for r in terms]);reject(lambda:p.start(0,0,True));assert p.drive is None
        p.start(0,0);reject(lambda:p.start(0,0));reject(lambda:p.read(0))
        p.advance(200e-9);assert p.state=='releasing';reject(lambda:p.read(0))
        p.advance(10e-6);first=p.read(0);assert first['decision']==expected
        first['samples'][0][0]=999;assert p.read(0)['samples'][0][0]!=999
        p.start(p.time,0);p.advance(p.time+10e-6);assert p.read(0)['decision']==expected
        p.abort(p.time);reject(lambda:p.read(0));reject(lambda:p.read(1));assert p.drive is None
        rows.append(dict(terms_ohm=terms,decision=expected,repeated=True))
    # Abort during stimulus: no reset of either capacitor; passive decay continues.
    p=Probe([load(75),load(1e8)]);p.start(0,0);p.advance(73e-9)
    before=[leg.x.copy() for leg in p.legs];p.abort(p.time)
    assert all((a==leg.x).all() for a,leg in zip(before,p.legs))
    p.advance(1e-6);assert p.state=='aborted' and p.result is None and p.drive is None
    report=dict(status='passed',connected=False,physical_qualification=False,cases=rows,
        abort_preserves_capacitors=True,release_required_before_result=True,
        source_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        dependency_hashes={n:hashlib.sha256(Path(__file__).with_name(n).read_bytes()).hexdigest() for n in ('release_screen.py','load_screen.py')},
        limitations=['Staged standalone controller, not yet connected to real chip scheduler, queues, management or supply loading.',
        'Three nominal two-leg cases; earlier load/sensor uncertainty tests do not automatically qualify the complete sequence.',
        'Post-abort recovery/rearm policy remains to be integrated; abort deliberately does not clear analog state.',
        'Active baseline restoration needs a physical driver and release-current budget.'])
    Path('/home/ubuntu/svalbard/projects/programmable_transceiver_platform/evidence/receiver-detect-sequence-screen.json').write_text(json.dumps(report,indent=2)+'\n')
    print('Passed two-leg present/absent/mixed sequences, repeated probes, busy rejection and capacitor-preserving abort')

if __name__=='__main__':main()
