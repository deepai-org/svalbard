import json
from pathlib import Path
from trim_calibration import OffsetTrimPlant,Calibration
rows=[]
for offset in (-.2,-.03,0.,.07,.2,.3,-.3):
    p=OffsetTrimPlant(offset)
    result=Calibration(p.write,p.above).run(0.)
    residual=p.apply(result['done_at']+1e-6,0.)
    if abs(offset)<.25:
        assert not result['range_limited'] and abs(residual)<=p.span/4095+1e-9
        # Independent held-out signal must receive the same correction.
        assert abs(p.apply(p.time,.123)-.123)<=p.span/4095+1e-9
    else:assert result['range_limited'] and abs(residual)>.049
    rows.append(dict(offset_v=offset,code=result['code'],residual_v=residual,range_limited=result['range_limited']))
# Comparator bias cannot be corrected merely by trusting the same comparator.
p=OffsetTrimPlant(.07,comparator_offset_v=.002)
r=Calibration(p.write,p.above).run(0.);residual=p.apply(r['done_at']+1e-6,0.)
assert abs(residual)>.0018
rows.append(dict(comparator_offset_v=.002,residual_v=residual,independent_verification_required=True))
report=dict(status='passed',cases=rows,chip_connected=False,physical_qualification=False,
 limitations=['Actuator range/monotonicity/settling are assumed; not PDK characterized.',
 'Shared-domain quiet-window and timed resource integration still required.',
 'Comparator-only convergence does not establish unbiased residual accuracy.'])
Path('/tmp/svalbard-calibration-pass877/report.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
