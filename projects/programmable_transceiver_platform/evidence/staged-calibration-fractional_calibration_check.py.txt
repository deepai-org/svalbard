"""Compose the staged maintenance controller with both pulse PLLs/fractional RF."""
import json,math
from pathlib import Path
from automatic_observation import AutomaticCalibrationChip
from fractional_rf_chip import FractionalRFChip
from managed_resources import command
from chip_model import decode_iq
class Candidate(AutomaticCalibrationChip,FractionalRFChip):
 pass
rows=[]
for mode,target in ((0,2412e6),(1,2437e6)):
 c=Candidate(observation_error_bound_v=.0003,adc_latency_s=30e-9,watchdog_s=1e-3)
 assert command(c,'cal_start',48)['accepted'];c.advance(c.time+40e-6)
 assert c.cal.valid
 c.configure_rf_carrier(int(target))
 c.configure_rx('external_tone',1,5e6,5e6,.2+0j,target-2.4e9)
 c.configure(mode,c.time);c.advance(c.time+40e-6)
 assert c.state=='active',(mode,c.state,c.events)
 start=c.time+100e-9;c.capture(8,start,gain=1)
 c.advance(start+3e-6);c.host_decoder.finish()
 assert c.host_samples==c.adc_words and len(c.host_samples)==8
 values=[decode_iq(w,c.bits) for w in c.host_samples]
 assert all(.15<abs(v)<.25 for v in values)
 rows.append(dict(mode=mode,target_hz=target,trim_code=c.trim.code,
     calibration_result=c.cal.result,maintenance=c.maintenance_accounting(),
     samples=len(values),magnitude_range=[min(map(abs,values)),max(map(abs,values))],
     rf_clock_class=type(c.rf_pll).__name__))
Path('/tmp/svalbard-calibration-pass877/fractional-calibration-report.json').write_text(json.dumps(dict(status='passed',cases=rows,complete_architecture=False,physical_qualification=False,limitations=['Narrow two-mode smoke check, not full lifecycle/quality regression','Calibration uncertainty remains explicitly assumed']),indent=2)+'\n')
print(rows)
