import sys,math
sys.path.insert(0,'/home/ubuntu/svalbard/projects/programmable_transceiver_platform/system_model/connected')
from programmable_chip import ProgrammableChip
from timed_calibration import TimedCalibration
from trim_calibration import OffsetTrimPlant

class ManagedCalibrationChip(ProgrammableChip):
    TILE_COMMANDS=ProgrammableChip.TILE_COMMANDS+('cal_start','cal_status','cal_abort','cal_trim')
    def __init__(self,trim_offsets_v=(.07,0.),**kwargs):
        if len(trim_offsets_v)!=2:raise ValueError('Require I and Q target offsets')
        self.trim_targets=[OffsetTrimPlant(v,comparator_offset_v=.002) for v in trim_offsets_v]
        self.trim_target=0;self.trim=self.trim_targets[0]
        self.cal=TimedCalibration(lambda t,c:self.trim.write(t,c),lambda t,p:self.trim.above(t,p))
        super().__init__(**kwargs)
    def quiet(self):
        return self.reference and self.state=='reset' and not self.session.armed and not self.adc_left and not self.adc_pending and not self.monitor_route
    def _require_target_free(self):
        if self.cal.busy:raise ValueError('Calibration owns analog target')
    def configure_rx(self,*args,**kwargs):
        self._require_target_free();return super().configure_rx(*args,**kwargs)
    def configure_filters(self,*args,**kwargs):
        self._require_target_free();return super().configure_filters(*args,**kwargs)
    def configure_rf_input(self,*args,**kwargs):
        self._require_target_free();return super().configure_rf_input(*args,**kwargs)
    def configure_lo(self,*args,**kwargs):
        self._require_target_free();return super().configure_lo(*args,**kwargs)
    def configure_rf_carrier(self,*args,**kwargs):
        self._require_target_free();return super().configure_rf_carrier(*args,**kwargs)
    def select_diagnostic(self,*args,**kwargs):
        self._require_target_free();return super().select_diagnostic(*args,**kwargs)
    def capture(self,*args,**kwargs):
        self._require_target_free();return super().capture(*args,**kwargs)
    def schedule(self,*args,**kwargs):
        self._require_target_free();return super().schedule(*args,**kwargs)
    def set_reference(self,present,time):
        result=super().set_reference(present,time)
        if not present and self.cal.busy:self.cal.cancel(time,'reference loss')
        return result
    def configure(self,mode,time):
        if self.cal.busy:raise ValueError('Calibration owns analog target')
        return super().configure(mode,time)
    def execute_management(self,operation,payload,time):
        if operation=='resource_count':
            if payload:raise ValueError('Reserved resource-count payload')
            return dict(value=9)
        if operation=='resource_status' and (payload==8 or payload==0 and self.cal.busy):
            owner=9 if self.cal.busy else 0
            return dict(value=owner|(int(self.cal.busy)<<8)|(int(bool(owner))<<10))
        if operation=='cal_start':
            if payload>>17 or not payload&65535:raise ValueError('Require 16-bit spacing and I/Q target bit')
            self._require_target_free()
            if not self.quiet():raise ValueError('Quiet ownership required')
            self.trim_target=payload>>16;self.trim=self.trim_targets[self.trim_target]
            self.cal.start(time,epoch=self.epoch,quiet=True,code=self.trim.code,
                           spacing=(payload&65535)*self.control_period,tolerance=.001)
            return {}
        if operation=='cal_abort':
            if payload:raise ValueError('Reserved calibration abort payload')
            self.cal.cancel(time);return {}
        if operation=='cal_trim':
            if payload not in (0,1):raise ValueError('Invalid trim target')
            return dict(value=self.trim_targets[payload].code)
        if operation=='cal_status':
            if payload:raise ValueError('Reserved calibration status payload')
            states=('idle','search','settle','observe','done','cancelled')
            result=self.cal.result or {}
            return dict(value=states.index(self.cal.state)|(int(self.cal.busy)<<8)|(int(self.cal.valid)<<9)
                        |(int(result.get('range_limited',False))<<10)
                        |(int(result.get('accuracy')=='failed')<<11)
                        |(int(result.get('accuracy')=='unverified')<<12)|(self.trim_target<<16))
        if self.cal.busy and operation in ('configure_rx','tile_configure','tile_discharge','diagnostic_select','monitor_select'):
            raise ValueError('Calibration owns analog target')
        return super().execute_management(operation,payload,time)
    def quiesce(self,time,reason):
        self.cal.cancel(time,reason)
        return super().quiesce(time,reason)
    def advance(self,time):
        if not math.isfinite(time) or time<self.time:raise ValueError('Invalid calibration time')
        while self.time<time:
            event=self.cal.next_event if self.cal.next_event is not None else math.inf
            command=self.command_events[0][0] if self.command_events else math.inf
            end=min(time,event,command)
            super().advance(end)
            self.cal.advance(self.time,epoch=self.epoch,quiet=self.quiet())
        super().advance(time)
        self.cal.advance(self.time,epoch=self.epoch,quiet=self.quiet())
