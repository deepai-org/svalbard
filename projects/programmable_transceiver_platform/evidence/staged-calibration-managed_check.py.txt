import json
from pathlib import Path
from managed_calibration import ManagedCalibrationChip
from managed_resources import command
c=ManagedCalibrationChip(watchdog_s=1e-3)
assert command(c,'cal_start',480)['accepted']
assert command(c,'cal_status')['value']&256
assert not command(c,'configure_mode',0)['accepted']
assert not command(c,'monitor_select',1)['accepted']
assert not command(c,'cal_start',480)['accepted']
c.advance(c.time+310e-6)
assert c.cal.state=='observe' and len(c.cal.history)==24
assert not c.cal.valid
assert command(c,'cal_abort')['accepted'] and c.trim.code==2048
assert command(c,'configure_mode',0)['accepted']
assert not command(c,'cal_start',480)['accepted']
report=dict(status='passed',comparisons=len(c.cal.history),final_state=c.state,
 controls=['serialized start/status/abort','busy mode rejection','busy monitor rejection','second start rejection','independent observation required','abort rollback','armed start rejection'],
 limitations=['ADC observation not yet scheduled by controller','quiet owner omits a full physical bias/route arbiter','fixed target offset and provisional 1mV tolerance are test fixtures'])
Path('/tmp/svalbard-calibration-pass877/managed-report.json').write_text(json.dumps(report,indent=2)+'\n')
print(report)
