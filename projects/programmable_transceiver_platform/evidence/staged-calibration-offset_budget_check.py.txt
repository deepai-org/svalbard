"""Translate residual DC into waveform error without subtracting DC in the fit."""
import sys,json,math
from pathlib import Path
sys.path.insert(0,'/home/ubuntu/svalbard/projects/programmable_transceiver_platform/system_model/connected')
from rf_quality_screen import quality
from chip_model import encode_iq,decode_iq
rows=[]
for amplitude in (.01,.03,.1,.3):
 reference=[amplitude*complex(math.cos(2*math.pi*k/64),math.sin(2*math.pi*k/64)) for k in range(1024)]
 for offset in (0.,.5/4095,.0005,.002,.005):
  for bits in (8,12):
   measured=[decode_iq(encode_iq(z+complex(offset,offset),bits),bits) for z in reference]
   q=quality(reference,measured)
   # Full periods make DC orthogonal to the training tone. Unquantized result
   # must match the analytical sqrt(2)*offset/amplitude law.
   ideal=quality(reference,[z+complex(offset,offset) for z in reference])
   expected=math.sqrt(2)*offset/amplitude
   assert abs(ideal['corrected_relative_rms']-expected)<1e-12
   rows.append(dict(amplitude_v=amplitude,offset_per_iq_v=offset,bits=bits,quality=q,ideal_relative_rms=expected))
report=dict(status='passed',cases=rows,complete_architecture=False,
 assumptions='Steady coherent tone; equal static I/Q offsets added immediately before ideal ADC; no clipping, noise, blocker, or gain uncertainty. 10% is the existing provisional screen, not a standard compliance limit.',
 conclusion='A one-trim-LSB residual is not itself an RF system requirement. Required offset depends on minimum post-gain signal amplitude and error allocation.')
Path('/tmp/svalbard-calibration-pass877/offset-budget-report.json').write_text(json.dumps(report,indent=2)+'\n')
for r in rows:
 if r['offset_per_iq_v']==.002 and r['bits']==12:print(r['amplitude_v'],r['quality']['corrected_relative_rms'],r['quality']['screen_pass'])
