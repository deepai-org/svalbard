import json
from pathlib import Path
from trim_calibration import OffsetTrimPlant,calibrate_reversed
rows=[]
for offset in (-.2,-.03,0.,.07,.2):
 for bias in (-.002,0,.002):
    p=OffsetTrimPlant(offset,comparator_offset_v=bias)
    r=calibrate_reversed(p,0.);residual=p.apply(r['done_at']+1e-6,0.)
    assert not r['range_limited'] and abs(residual)<=p.span/4095+1e-9
    rows.append(dict(offset_v=offset,comparator_bias_v=bias,code=r['code'],residual_v=residual))
p=OffsetTrimPlant(.249,comparator_offset_v=.002);old=p.code
r=calibrate_reversed(p,0.)
assert r['range_limited'] and p.code==old
Path('/tmp/svalbard-calibration-pass877/reversal-report.json').write_text(json.dumps(dict(status='passed',cases=rows,
 range_failure_restores_code=True,chip_connected=False,physical_qualification=False,
 limitations=['Ideal polarity reversal assumes invariant comparator bias and no switch-induced offset.',
 'ADC verification, gain/nonlinearity, temperature drift and resource arbitration remain open.']),indent=2)+'\n')
print('Passed15 polarity-reversal calibration cases and out-of-range rollback')
